% --- Custom Command: Centered Section Header with Full-Width Rule ---
\newcommand{\cvsection}[1]{%
    \vspace{9pt}%
    \begin{center}%
        {\fontsize{11.5}{14}\selectfont \textbf{#1}}\par
        \vspace{3pt}%
        \color{rulecolor}\rule{\textwidth}{0.5pt}%
    \end{center}%
    \vspace{-2pt}%
}

% --- Custom Command: Experience Entry ---
% #1: Company Name
% #2: Position / Job Title
% #3: Location
% #4: Dates
% #5: Overview / Scope description
% #6: Bullet points
\newcommand{\experienceentry}[6]{%
    \vspace{6pt}%
    \noindent{\fontsize{10.5}{13}\selectfont \textbf{#1}}\par
    \vspace{1.5pt}%
    \noindent{\fontsize{9.5}{12}\selectfont \color{metacolor} #2 \textbar{} #3 \textbar{} #4}\par
    \vspace{2.5pt}%
    \noindent{\fontsize{9}{11.5}\selectfont \color{bodydark} #5}\par
    \vspace{1.5pt}%
    #6%
}

% --- Custom Command: Skills Card Box (2x2 Grid) ---
% #1: Category Title
% #2: Skills List / Description
\newcommand{\skillcard}[2]{%
    \begin{tcolorbox}[
        colback=white,
        colframe=boxborder,
        boxrule=0.6pt,
        arc=1.5pt,
        top=4.5pt,
        bottom=4.5pt,
        left=6pt,
        right=6pt,
        width=0.488\textwidth,
        height=36pt,
        valign=top,
        nobeforeafter
    ]
        {\fontsize{9}{11.5}\selectfont \textbf{\color{boxtitle} #1}}\par
        \vspace{2pt}
        {\fontsize{8}{10.5}\selectfont \color{boxtext} #2}
    \end{tcolorbox}%
}

% --- Custom Command: Education Entry ---
% #1: Institution Name

}
